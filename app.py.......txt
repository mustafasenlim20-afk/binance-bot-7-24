# ===================== TEKİL KOPYA KİLİDİ (LOCK) =====================
LOCK_FILE = "bot.lock"
lock_fd = None


def acquire_lock():
    global lock_fd
    lock_fd = open(LOCK_FILE, "a+")
    try:
        if os.name == 'nt':
            import msvcrt
            msvcrt.locking(lock_fd.fileno(), msvcrt.LK_NBLCK, 1)
        else:
            import fcntl
            fcntl.flock(lock_fd.fileno(), fcntl.LOCK_EX | fcntl.LOCK_NB)
        lock_fd.seek(0)
        lock_fd.truncate()
        lock_fd.write(str(os.getpid()))
        lock_fd.flush()
    except (IOError, OSError) as e:
        if e.errno in (errno.EACCES, errno.EAGAIN):
            log.error("⛔ KRİTİK: Botun başka bir kopyası zaten çalışıyor! Bu kopya kapatılıyor.")
            sys.exit(0)
        raise

# ---------------- BAN / HIZ KORUMASI ----------------
ban_until = 0.0
ban_lock = threading.Lock()
ban_strikes = 0            # ust uste kac kez limit yendi (ustel bekleme icin)
last_ban_ts = 0.0
BAN_FILE = "ban_state.json"


def _save_ban():
    """Ban suresini dosyaya yazar: bot yeniden baslarsa banli IP'ye hemen istek atip cezayi uzatmasin."""
    try:
        with open(BAN_FILE, "w") as f:
            json.dump({"until": ban_until}, f)
    except Exception:
        pass


def _load_ban():
    global ban_until
    try:
        with open(BAN_FILE) as f:
            ban_until = max(ban_until, float(json.load(f).get("until", 0)))
        if ban_until > time.time():
            log.warning(f"Onceki ban suruyor, {int(ban_until - time.time())} sn istek atilmayacak.")
    except Exception:
        pass


def note_ban(err):
    global ban_until, ban_strikes, last_ban_ts
    text = str(err)
    now = time.time()
    wait_until = 0.0
    m = re.search(r"banned until (\d{10,})", text)
    if m:
        wait_until = int(m.group(1)) / 1000.0 + 15
    elif "418" in text or "-1003" in text or "429" in text or "Too many requests" in text:
        with ban_lock:
            if now >= ban_until:                  # zaten beklemedeyken gelen gec hatalar sayilmaz
                if now - last_ban_ts > 900:       # 15 dk sakin gectiyse sayac sifirlanir
                    ban_strikes = 0
                ban_strikes = min(ban_strikes + 1, 5)
                last_ban_ts = now
            n = max(ban_strikes, 1)
        wait_until = now + min(120 * 2 ** (n - 1), 1800)    # 2, 4, 8, 16, 30 dk
    elif "restricted location" in text or "451" in text:
        wait_until = now + 600
    if wait_until:
        with ban_lock:
            if wait_until > ban_until:
                ban_until = wait_until
                left = int(ban_until - time.time())
                log.error(f"⛔ Binance yasagi/limit algilandi, {left} sn istek atilmayacak.")
                _save_ban()
        return True
    return False


def wait_if_banned():
    while True:
        left = ban_until - time.time()
        if left <= 0:
            return
        time.sleep(min(left, 60))

def journal_close(symbol, st, reason):
    """Kapanan pozisyonun gerceklesen kar/zararini borsadan (tek istek) okuyup gunluge yazar. REST: kilidin disinda cagir."""
    if not st:
        return
    opened_at = float(st.get("opened_at") or time.time())
    pnl = fee = net = None
    try:
        wait_if_banned()
        trades = ex.fetch_my_trades(symbol, since=int(opened_at * 1000) - 60_000, limit=100)
        pnl = 0.0
        fee = 0.0
        for t in trades:
            info = t.get("info") or {}
            pnl += float(info.get("realizedPnl") or 0)
            fee += float((t.get("fee") or {}).get("cost") or 0)
        net = pnl - fee
    except Exception as e:
        log.warning(f"{symbol} gunluk icin islem gecmisi okunamadi: {e}")
        note_ban(e)
    journal("close", symbol, side=st.get("side"), lev=st.get("lev"), entry=st.get("entry"),
            pnl_usdt=round(pnl, 4) if pnl is not None else "",
            fee_usdt=round(fee, 4) if fee is not None else "",
            net_usdt=round(net, 4) if net is not None else "",
            hold_min=round((time.time() - opened_at) / 60), reason=reason)

class MiniWS:
    """Kucuk RFC 6455 websocket istemcisi (ws:// ve wss://). Sunucuya paket kurmaya gerek birakmaz."""
    GUID = b"258EAFA5-E914-47DA-95CA-C5AB0DC85B11"
    MAX_FRAME = 32 * 1024 * 1024

    def __init__(self, url, proxy_url=None, ssl_context=None, connect_timeout=15.0, read_timeout=1.0):
        self.url = url
        self.proxy_url = proxy_url
        self.ssl_context = ssl_context
        self.connect_timeout = connect_timeout
        self.read_timeout = read_timeout
        self.sock = None
        self.buf = bytearray()
        self.send_lock = threading.Lock()
        self.frag_data = bytearray()

    @staticmethod
    def _read_http_head(sock):
        data = bytearray()
        while b"\r\n\r\n" not in data:
            chunk = sock.recv(4096)
            if not chunk:
                raise ConnectionError("sunucu HTTP basligi gelmeden baglantiyi kapatti")
            data += chunk
            if len(data) > 65536:
                raise ConnectionError("HTTP basligi cok buyuk")
        head, _, rest = bytes(data).partition(b"\r\n\r\n")
        return head, rest

    @staticmethod
    def _xor(data, mask):
        n = len(data)
        if n == 0:
            return b""
        full = (mask * (n // 4 + 1))[:n]
        return (int.from_bytes(data, "big") ^ int.from_bytes(full, "big")).to_bytes(n, "big")

    def connect(self):
        u = urlparse(self.url)
        secure = u.scheme == "wss"
        host = u.hostname
        port = u.port or (443 if secure else 80)
        path = (u.path or "/") + (("?" + u.query) if u.query else "")

        proxy = urlparse(self.proxy_url) if self.proxy_url else None
        if proxy is not None and (not proxy.hostname or proxy.scheme.startswith("socks")):
            proxy = None        # socks desteklenmiyor: dogrudan baglanmayi dene

        if proxy is not None:
            sock = socket.create_connection((proxy.hostname, proxy.port or 8080), timeout=self.connect_timeout)
            req = f"CONNECT {host}:{port} HTTP/1.1\r\nHost: {host}:{port}\r\n"
            if proxy.username:
                cred = f"{unquote(proxy.username)}:{unquote(proxy.password or '')}".encode()
                req += "Proxy-Authorization: Basic " + base64.b64encode(cred).decode() + "\r\n"
            sock.sendall((req + "\r\n").encode())
            try:
                head, _ = self._read_http_head(sock)
            except Exception:
                sock.close()
                raise
            status = head.split(b"\r\n", 1)[0].decode("latin1")
            if len(status.split()) < 2 or status.split()[1] != "200":
                sock.close()
                raise ConnectionError(f"proxy CONNECT reddetti: {status}")
        else:
            sock = socket.create_connection((host, port), timeout=self.connect_timeout)

        try:
            if secure:
                ctx = self.ssl_context or ssl.create_default_context()
                sock = ctx.wrap_socket(sock, server_hostname=host)
            try:
                sock.setsockopt(socket.IPPROTO_TCP, socket.TCP_NODELAY, 1)
            except OSError:
                pass
            key = base64.b64encode(os.urandom(16)).decode()
            host_hdr = host if port in (80, 443) else f"{host}:{port}"
            sock.sendall((
                f"GET {path} HTTP/1.1\r\nHost: {host_hdr}\r\nUpgrade: websocket\r\n"
                f"Connection: Upgrade\r\nSec-WebSocket-Key: {key}\r\n"
                f"Sec-WebSocket-Version: 13\r\nUser-Agent: Mozilla/5.0\r\n\r\n"
            ).encode())
            head, rest = self._read_http_head(sock)
            lines = head.decode("latin1").split("\r\n")
            status = lines[0]
            parts = status.split()
            if len(parts) < 2 or parts[1] != "101":
                body = rest[:200].decode("utf-8", "replace").strip()
                raise ConnectionError(f"el sikisma reddedildi: {status} {body}")
            headers = {}
            for ln in lines[1:]:
                if ":" in ln:
                    k, v = ln.split(":", 1)
                    headers[k.strip().lower()] = v.strip()
            expect = base64.b64encode(hashlib.sha1(key.encode() + self.GUID).digest()).decode()
            if headers.get("sec-websocket-accept") != expect:
                raise ConnectionError("el sikisma dogrulanamadi (Sec-WebSocket-Accept)")
            self.buf = bytearray(rest)      # basliktan sonra gelmis olabilecek ilk cerceve baytlari
            sock.settimeout(self.read_timeout)
            self.sock = sock
        except Exception:
            try:
                sock.close()
            except Exception:
                pass
            raise

    def _fill(self):
        try:
            chunk = self.sock.recv(65536)
        except socket.timeout:
            return False
        except ssl.SSLWantReadError:
            return False
        if not chunk:
            raise ConnectionError("sunucu baglantiyi kapatti")
        self.buf += chunk
        return True

    def _frame(self):
        b = self.buf
        if len(b) < 2:
            return None
        b1, b2 = b[0], b[1]
        fin = bool(b1 & 0x80)
        op = b1 & 0x0F
        masked = bool(b2 & 0x80)
        ln = b2 & 0x7F
        pos = 2
        if ln == 126:
            if len(b) < 4:
                return None
            ln = struct.unpack(">H", bytes(b[2:4]))[0]
            pos = 4
        elif ln == 127:
            if len(b) < 10:
                return None
            ln = struct.unpack(">Q", bytes(b[2:10]))[0]
            pos = 10
        mask = b""
        if masked:
            if len(b) < pos + 4:
                return None
            mask = bytes(b[pos:pos + 4])
            pos += 4
        if ln > self.MAX_FRAME:
            raise ConnectionError("cerceve cok buyuk")
        if len(b) < pos + ln:
            return None
        payload = bytes(b[pos:pos + ln])
        del b[:pos + ln]
        if masked:
            payload = self._xor(payload, mask)
        return fin, op, payload

    def _send(self, op, payload=b""):
        n = len(payload)
        hdr = bytearray([0x80 | op])
        if n < 126:
            hdr.append(0x80 | n)
        elif n < 65536:
            hdr.append(0x80 | 126)
            hdr += struct.pack(">H", n)
        else:
            hdr.append(0x80 | 127)
            hdr += struct.pack(">Q", n)
        mask = os.urandom(4)
        data = bytes(hdr) + mask + self._xor(payload, mask)
        with self.send_lock:
            self.sock.sendall(data)

    def send_pong(self):
        self._send(0xA, b"")

    def send_text(self, text):
        self._send(0x1, text.encode())

    def recv(self):
        """Tam bir mesaj (metin) dondurur. Veri yoksa ~1 sn sonra None doner."""
        while True:
            fr = self._frame()
            if fr is None:
                if not self._fill():
                    return None
                continue
            fin, op, payload = fr
            if op == 0x9:               # ping -> pong
                self._send(0xA, payload)
                continue
            if op == 0xA:               # pong
                continue
            if op == 0x8:               # close
                try:
                    self._send(0x8, payload[:2])
                except Exception:
                    pass
                raise ConnectionError("sunucu baglantiyi kapatti (close)")
            if op in (0x1, 0x2):
                self.frag_data = bytearray(payload)
            elif op == 0x0:
                self.frag_data += payload
            else:
                continue
            if fin:
                data = bytes(self.frag_data)
                self.frag_data = bytearray()
                return data.decode("utf-8", "replace")

    def close(self):
        s = self.sock
        self.sock = None
        if s is None:
            return
        try:
            self.sock = s
            self._send(0x8, b"")
        except Exception:
            pass
        self.sock = None
        try:
            s.close()
        except Exception:
            pass

    return f"{raw[:-4]}/USDT:USDT"

def wait_for_ws(timeout=30):
    """Bot yeni acildiysa websocket'in veri almasini bekler (ilk tarama bos yere REST yedege dusmesin)."""
    if time.time() - START_TS > 300:
        return
    t0 = time.time()
    while time.time() - t0 < timeout:
        if ws_healthy() and first_tick_ts and time.time() - first_tick_ts >= 8:
            return
        time.sleep(1)


def ws_on_message(text):
    global tickers_ts, first_tick_ts
    try:
        msg = json.loads(text)
    except Exception:
        return
    if not isinstance(msg, dict):
        return
    data = msg.get("data")
    if data is None:       # abonelik cevabi vb.
        return
    stream = msg.get("stream", "")

    if stream == "!miniTicker@arr":
        now = time.time()
        with ws_lock:
            for it in data:
                try:
                    raw = it["s"]
                    price = float(it["c"])
                    vol = float(it["q"])
                except (KeyError, ValueError, TypeError):
                    continue
                tickers[raw] = {"price": price, "vol": vol}
                h = tick_hist.get(raw)
                if h is None:
                    h = tick_hist[raw] = deque(maxlen=HIST_MAXLEN)
                if not h or now - h[-1][0] >= HIST_SAMPLE_SECONDS:
                    h.append((now, price, vol))
            tickers_ts = now
            if not first_tick_ts:
                first_tick_ts = now

    elif "@kline_" in stream:
        try:
            k = data["k"]
            raw = data["s"]
            row = [int(k["t"]), float(k["o"]), float(k["h"]), float(k["l"]),
                   float(k["c"]), float(k["v"]), bool(k["x"])]
        except (KeyError, ValueError, TypeError):
            return
        with ws_lock:
            arr = klines.setdefault(raw, [])
            if not arr:
                arr.append(row)
            elif arr[-1][0] == row[0]:
                arr[-1] = row
            elif row[0] > arr[-1][0]:
                if row[0] - arr[-1][0] > TF_MS * 1.5:
                    dirty.add(raw)           # arada mum kacti, REST ile yeniden yuklenecek
                arr[-1][6] = True            # yeni mum geldiyse onceki kapanmistir
                arr.append(row)
                if len(arr) > KLINE_KEEP:
                    del arr[0]
            kline_ts[raw] = time.time()


def ws_build_url(base):
    """Binance 'birlesik akis' adresi: fiyat akisi + istenen coinlerin mum akislari."""
    with ws_lock:
        syms = sorted(subscribed)
    streams = ["!miniTicker@arr"] + [f"{r.lower()}@kline_{TIMEFRAME}" for r in syms]
    return f"{base}?streams={'/'.join(streams)}", set(syms)


def ws_thread():
    """Surekli calisir; kopunca ya da yeni coin eklenince kendini yeniden baglar."""
    global ws_connected
    idx = 0
    fails = 0
    while True:
        base = WS_URLS[idx % len(WS_URLS)]
        url, conn_syms = ws_build_url(base)
        started = time.time()
        got_data = False
        client = None
        try:
            client = MiniWS(url, proxy_url=PROXY_URL)
            client.connect()
            ws_connected = True
            log.info(f"✅ WebSocket baglandi: {base} (1 fiyat akisi + {len(conn_syms)} mum akisi)")
            last_rx = time.time()
            last_pong = last_rx
            while True:
                msg = client.recv()
                now = time.time()
                if msg is not None:
                    last_rx = now
                    if not got_data:
                        got_data = True
                        log.info("📡 WebSocket veri akisi basladi.")
                    ws_on_message(msg)
                elif now - last_rx > WS_IDLE_RECONNECT:
                    raise ConnectionError(f"{WS_IDLE_RECONNECT} sn boyunca veri gelmedi")
                if now - last_pong > 60:
                    client.send_pong()       # istenmemis pong'a Binance izin veriyor (baglantiyi canli tutar)
                    last_pong = now
                with ws_lock:
                    missing = len(subscribed - conn_syms)
                if missing and now - started > WS_RESUB_MIN_AGE:
                    log.info(f"WebSocket yeni coin akislari icin yeniden baglaniyor (+{missing} coin).")
                    break
                if now - started > 23 * 3600:      # Binance baglantiyi 24 saatte keser
                    break
        except Exception as e:
            log.error(f"WebSocket hatasi ({base}): {type(e).__name__}: {e}")
        finally:
            ws_connected = False
            if client is not None:
                client.close()

        if got_data:
            fails = 0
            time.sleep(1)
        else:
            fails += 1
            idx += 1                                # diger adresi dene
            time.sleep(min(3 * fails, 30))

def is_tradfi(symbol):

    try:

        utype = str(info.get("underlyingType", "")).upper()
        subs = [str(x).upper() for x in (info.get("underlyingSubType") or [])]

            return True

    except Exception:
        return False

def clamp(x, lo, hi):
    return max(lo, min(hi, x))


def prescreen_movers(symbols):
    """Sadece WS fiyat gecmisinden (REST YOK) son PRESCREEN_WINDOW sn'de en hizli hareket eden coinleri bulur.
    Her coin icin: fiyat degisimi % ve hacim hizi (pencere hacmi / 24s ortalamanin ayni sureye dusen payi).
    Isinma bitmediyse (yeterli gecmis yok) None doner; bitmis ama kimse esigi gecmediyse bos liste doner.
    Not: 24s hacim 'kayan pencere' oldugu icin pencere hacmi hafif eksik hesaplanir; sadece siralama icindir."""
    with ws_lock:
        hist = {s: list(tick_hist.get(to_raw(s), ())) for s in symbols}
    ready = 0
    out = []
    for s in symbols:
        h = hist[s]
        if len(h) < 2:
            continue
        t_now, p_now, v_now = h[-1]
        target = t_now - PRESCREEN_WINDOW
        ref = None
        for smp in h:                           # en eskiden yeniye: pencerenin icindeki ilk ornek
            if smp[0] >= target:
                ref = smp
                break
        if ref is None:
            continue
        span = t_now - ref[0]
        if span < PRESCREEN_WARMUP or ref[1] <= 0:
            continue
        ready += 1
        chg = (p_now / ref[1] - 1.0) * 100.0
        base = v_now * span / 86400.0
        ratio = (v_now - ref[2]) / base if base > 0 else 0.0
        if abs(chg) < PRESCREEN_MIN_MOVE_PCT:
            continue
        heat = abs(chg) * clamp(ratio, 0.5, 3.0)
        out.append({"symbol": s, "chg": chg, "ratio": ratio, "heat": heat})
    if ready == 0:
        return None
    out.sort(key=lambda m: m["heat"], reverse=True)
    return out[:PRESCREEN_TOP]

def window_change_pct(raw):
    """Bir coinin son BTC_WINDOW sn'deki % degisimi (sadece WS fiyat gecmisi, REST YOK).
    Yeterli gecmis yoksa None doner."""
    with ws_lock:
        h = list(tick_hist.get(raw, ()))
    if len(h) < 2:
        return None
    t_now, p_now, _ = h[-1]
    target = t_now - BTC_WINDOW
    ref = None
    for smp in h:
        if smp[0] >= target:
            ref = smp
            break
    if ref is None or ref[1] <= 0 or t_now - ref[0] < BTC_MIN_SPAN:
        return None
    return (p_now / ref[1] - 1.0) * 100.0


def btc_change_pct():
    """BTC'nin son BTC_WINDOW sn'deki % degisimi (sadece WS fiyat gecmisi, REST YOK).
    Yeterli gecmis yoksa None doner (filtre karar vermez)."""
    return window_change_pct("BTCUSDT")


def relative_strength(symbol, side, btc_chg):
    """Coinin BTC'ye gore gucu: (coin % - BTC %) yonun yonunde. Dondurur: (puan, fark_yuzde).
    Puan sadece pozitif fark icin verilir (BTC ile ayni gidiyorsa 0); veri yoksa (0, None). REST YOK. BTC islem gormez, sadece referans."""
    if not REL_STRENGTH_ENABLED or btc_chg is None:
        return 0.0, None
    coin_chg = window_change_pct(to_raw(symbol))
    if coin_chg is None:
        return 0.0, None
    diff = (coin_chg - btc_chg) * (1.0 if side == "long" else -1.0)
    pts = REL_STRENGTH_PTS * clamp(diff / REL_STRENGTH_FULL_PCT, 0.0, 1.0)
    return pts, diff


oi_cache = {}     # symbol -> (zaman, oi_degisim_yuzde veya None)


def oi_change_pct(symbol):
    """Son ~OI_BARS*OI_PERIOD icindeki Open Interest % degisimi. Tek REST istegi, OI_CACHE_SECONDS onbellekli,
    ban varken istek atmaz. Hata/veri yoksa None (fail-open: puan degismez)."""
    now = time.time()
    cached = oi_cache.get(symbol)
    if cached and now - cached[0] < OI_CACHE_SECONDS:
        return cached[1]
    if ban_until > now:
        return None
    chg = None
    try:
        rows = ex.fetch_open_interest_history(symbol, OI_PERIOD, None, OI_BARS)
        rows = sorted(rows or [], key=lambda r: r.get("timestamp") or 0)
        vals = []
        for r in rows:
            v = r.get("openInterestValue")
            if v is None:
                v = r.get("openInterestAmount")
            if v is not None:
                vals.append(float(v))
        if len(vals) >= 2 and vals[0] > 0:
            chg = (vals[-1] / vals[0] - 1.0) * 100.0
    except Exception as e:
        log.warning(f"{symbol} OI okunamadi (puan degismedi): {e}")
        note_ban(e)
    oi_cache[symbol] = (now, chg)
    return chg


# ---------------- Mum verisi ----------------
def is_fresh(raw):
    with ws_lock:
        arr = klines.get(raw)
        return bool(arr) and len(arr) >= 60 and raw not in dirty \
            and (time.time() - kline_ts.get(raw, 0)) < KLINE_STALE_SECONDS


def ensure_streams(symbols):
    """Yeni adaylarin mum akislarini kaydeder; websocket kendini yeniden baglayarak bunlari ekler.
    Yer kalmazsa en eski hareketli olmus (ve acik pozisyonu olmayan) coinlerin akislari cikarilir."""
    with trail_lock:
        protect = {to_raw(s) for s in trail_state}
    with ws_lock:
        new = [to_raw(s) for s in symbols if to_raw(s) not in subscribed]
        room = MAX_STREAMS - len(subscribed)
        if len(new) > room:
            need = len(new) - room
            cands = sorted((r for r in subscribed if r not in protect and r not in new),
                           key=lambda r: hot_ts.get(r, 0.0))
            for r in cands[:need]:
                subscribed.discard(r)
                klines.pop(r, None)
                kline_ts.pop(r, None)
                dirty.discard(r)
            room = MAX_STREAMS - len(subscribed)
        new = new[:max(0, room)]
        subscribed.update(new)
    if new:
        log.info(f"{len(new)} aday coin icin mum akisi isteniyor.")


def seed_symbol(symbol):
    """Gecmis mumlari REST'ten BIR KEZ yukler (sonrasi WS besler)."""
    raw = to_raw(symbol)
    ohlcv = ex.fetch_ohlcv(symbol, timeframe=TIMEFRAME, limit=SEED_LIMIT)
    rows = [[int(r[0]), r[1], r[2], r[3], r[4], r[5], True] for r in ohlcv]
    if rows:
        rows[-1][6] = False       # sonuncu mum henuz olusuyor
    with ws_lock:
        existing = klines.get(raw, [])
        if rows:
            last_t = rows[-1][0]
            same = [r for r in existing if r[0] == last_t]
            if same:
                rows[-1] = same[-1]                       # WS'teki surum daha tazedir
            rows = rows + [r for r in existing if r[0] > last_t]
        klines[raw] = rows[-KLINE_KEEP:]
        kline_ts[raw] = time.time()
        dirty.discard(raw)


def prepare_data(symbols, skip):
    ensure_streams(symbols)
    need = [s for s in symbols if s not in skip and not is_fresh(to_raw(s))]
    if not need:
        return
    total_need = len(need)
    need = need[:SEED_PER_CYCLE]      # en hareketli coinler once, kalanlar sonraki turda (istek patlamasi olmasin)
    log.info(f"Gecmis mum yukleniyor (REST, bir kerelik, aralikli): {len(need)}/{total_need} coin")
    for s in need:
        wait_if_banned()
        try:
            seed_symbol(s)
        except Exception as e:
            log.error(f"{s} gecmis mum hatasi: {e}")
            if note_ban(e):
                break
        time.sleep(SEED_SLEEP)


# ---------------- INDIKATORLER ----------------
def calculate_indicators(df):
    df["ema_fast"] = df["close"].ewm(span=9, adjust=False).mean()
    df["ema_slow"] = df["close"].ewm(span=21, adjust=False).mean()

    # MACD histogram (yeni): momentumun hizlanip yavaslamadigini gosterir
    macd_line = (df["close"].ewm(span=MACD_FAST, adjust=False).mean()
                 - df["close"].ewm(span=MACD_SLOW, adjust=False).mean())
    macd_sig = macd_line.ewm(span=MACD_SIGNAL, adjust=False).mean()
    df["macd_hist"] = macd_line - macd_sig

    delta = df["close"].diff()
    gain = delta.clip(lower=0).ewm(alpha=1 / 14, adjust=False).mean()
    loss = (-delta.clip(upper=0)).ewm(alpha=1 / 14, adjust=False).mean()
    rs = gain / loss
    df["rsi"] = 100 - (100 / (1 + rs))

    prev_close = df["close"].shift()
    tr = pd.concat([
        df["high"] - df["low"],
        (df["high"] - prev_close).abs(),
        (df["low"] - prev_close).abs(),
    ], axis=1).max(axis=1)
    df["atr"] = tr.rolling(14).mean()

    up = df["high"].diff()
    down = -df["low"].diff()
    plus_dm = up.where((up > down) & (up > 0), 0.0)
    minus_dm = down.where((down > up) & (down > 0), 0.0)
    tr_s = tr.ewm(alpha=1 / 14, adjust=False).mean()
    df["plus_di"] = 100 * plus_dm.ewm(alpha=1 / 14, adjust=False).mean() / tr_s
    df["minus_di"] = 100 * minus_dm.ewm(alpha=1 / 14, adjust=False).mean() / tr_s
    di_sum = (df["plus_di"] + df["minus_di"]).replace(0, float("nan"))
    dx = 100 * (df["plus_di"] - df["minus_di"]).abs() / di_sum
    df["adx"] = dx.ewm(alpha=1 / 14, adjust=False).mean()

    # ---------- ERKEN TREND INDIKATORLERI ----------
    # 1) Squeeze: Bollinger bandi Keltner'in icine girerse sikisma var (sq_on). Bitisi = hareketin baslangici.
    ma = df["close"].rolling(BB_PERIOD).mean()
    sd = df["close"].rolling(BB_PERIOD).std(ddof=0)
    kc_atr = tr.rolling(BB_PERIOD).mean()
    df["sq_on"] = ((ma + BB_MULT * sd) < (ma + KC_MULT * kc_atr)) & ((ma - BB_MULT * sd) > (ma - KC_MULT * kc_atr))

    # 2) Squeeze momentum (LazyBear): fiyatin orta cizgiye gore sapmasinin dogrusal regresyon ucu. Isaret = yon.
    mid = ((df["high"].rolling(BB_PERIOD).max() + df["low"].rolling(BB_PERIOD).min()) / 2.0 + ma) / 2.0
    df["sq_mom"] = (df["close"] - mid).rolling(BB_PERIOD).apply(_linreg_end, raw=True)

    # 3) Aroon: son N mumda en yuksek/en dusuk ne kadar yakin zamanda goruldu (0-100)
    n = AROON_PERIOD
    df["aroon_up"] = df["high"].rolling(n + 1).apply(np.argmax, raw=True) * (100.0 / n)
    df["aroon_dn"] = df["low"].rolling(n + 1).apply(np.argmin, raw=True) * (100.0 / n)

    # 4) Choppiness: yuksek = yatay/testere, dusuk = trend. Yuksekten dusmeye baslarsa yataydan trende gecis.
    rng = (df["high"].rolling(CHOP_PERIOD).max() - df["low"].rolling(CHOP_PERIOD).min()).replace(0, np.nan)
    df["chop"] = 100.0 * np.log10(tr.rolling(CHOP_PERIOD).sum() / rng) / np.log10(CHOP_PERIOD)
    return df


def _linreg_end(y):
    """Penceredeki degerlere dogru uydurur, dogrunun SON noktasini dondurur (squeeze momentum icin)."""
    n = len(y)
    x = np.arange(n, dtype=float)
    xm = x.mean()
    ym = y.mean()
    var = ((x - xm) ** 2).sum()
    if var == 0:
        return np.nan
    b = ((x - xm) * (y - ym)).sum() / var
    return ym - b * xm + b * (n - 1)


def get_df(symbol):
    """WebSocket onbelleginden DataFrame (REST istegi YOK)."""
    raw = to_raw(symbol)
    with ws_lock:
        rows = [r[:] for r in klines.get(raw, [])]
    if len(rows) < 2:
        return None
    df = pd.DataFrame(rows, columns=COLS)
    if not rows[-1][6]:          # olusmakta olan mumu at, sadece kapanmislar kalsin
        df = df.iloc[:-1]

def htf_df_from_ws(symbol):
    """1 saatlik mumlari, WS/onbellekteki 15 dk mumlardan URETIR (REST istegi YOK).
    Sadece tamamlanmis saatler kullanilir (her saatte TF'nin tum mumlari kapanmis olmali)."""
    raw = to_raw(symbol)
    with ws_lock:
        rows = [r[:] for r in klines.get(raw, [])]
    if len(rows) < 20:
        return None
    df = pd.DataFrame(rows, columns=COLS)
    df = df[df["closed"]].copy()
    if df.empty:
        return None
    per_hour = max(int(3_600_000 // TF_MS), 1)
    df["hour"] = df["timestamp"] // 3_600_000
    df = df[df.groupby("hour")["timestamp"].transform("count") >= per_hour]
    if df.empty:
        return None
    g = df.groupby("hour")
    h = pd.DataFrame({
        "timestamp": g["timestamp"].first(),
        "open": g["open"].first(),
        "high": g["high"].max(),
        "low": g["low"].min(),
        "close": g["close"].last(),
        "volume": g["volume"].sum(),
    }).reset_index(drop=True)
    if len(h) < 30:
        return None
    return calculate_indicators(h)

# ---------------- TREND PUANLAMA ----------------
def _trigger_states(df, side, atr):
    """Dort erken tetigin yone gore durumunu dondurur: (squeeze_cikisi, aroon_donusu, chop_dususu, taze_ema_kesisimi).
    Squeeze ve EMA tetikleri, fiyat tetik seviyesinden yonunde MAX_SINCE_TRIGGER_ATR'den fazla uzaklastiysa GECERSIZ sayilir
    (tetik eskimis, hareketin basi kacmis)."""
    n = len(df)
    close = df["close"]
    close_now = float(close.iloc[-1])

    def run_atr(ref_idx):
        """Tetik mumundan bu yana fiyatin SIGNAL yonunde kac ATR gittigi."""
        d = close_now - float(close.iloc[ref_idx])
        return (d if side == "long" else -d) / atr if atr > 0 else 0.0

    sq = df["sq_on"].astype(bool)
    last_sq_on = bool(sq.iloc[-1])
    recent_on = bool(sq.iloc[-1 - SQUEEZE_LOOKBACK:-1].any())
    squeeze_release = (not last_sq_on) and recent_on
    if squeeze_release:
        start = n - 1 - SQUEEZE_LOOKBACK
        w = sq.iloc[start:n - 1].to_numpy()
        j = start + int(np.flatnonzero(w).max())          # sikismanin son 'acik' oldugu mum = kirilim seviyesi
        if run_atr(j) > MAX_SINCE_TRIGGER_ATR:
            squeeze_release = False

    def aroon_ok(i):
        up, dn = float(df["aroon_up"].iloc[i]), float(df["aroon_dn"].iloc[i])
        if side == "long":
            return up >= AROON_STRONG and dn <= AROON_WEAK
        return dn >= AROON_STRONG and up <= AROON_WEAK

    # "donus": simdi saglaniyor ama 4 mum once saglanmiyordu (yani yeni dondu)
    aroon_turn = aroon_ok(-1) and not aroon_ok(-5)

    chop = df["chop"]
    was_choppy = float(chop.iloc[-1 - CHOP_LOOKBACK:-1].max()) >= CHOP_HIGH
    chop_falling = float(chop.iloc[-1]) < float(chop.iloc[-1 - CHOP_FALLING_BARS])
    chop_break = was_choppy and chop_falling

    # 4) TAZE EMA9/21 kesisimi: son EMA_CROSS_BARS mum icinde yone gecmis, simdi hala yonde, EMA21 yonunde egimli.
    # (Uzun suredir ayni yonde duran EMA durumu sayilmaz.)
    ef_s, es_s = df["ema_fast"], df["ema_slow"]
    side_state = (ef_s > es_s) if side == "long" else (ef_s < es_s)
    ema_cross = False
    if bool(side_state.iloc[-1]):
        win = side_state.iloc[-1 - EMA_CROSS_BARS:-1]
        if not bool(win.all()):
            start = n - 1 - EMA_CROSS_BARS
            k = start + int(np.flatnonzero(~win.to_numpy()).max()) + 1     # kesisimin gerceklestigi mum
            slope = float(es_s.iloc[-1]) - float(es_s.iloc[-1 - EMA_SLOPE_BARS])
            slope_ok = slope > 0 if side == "long" else slope < 0
            if slope_ok and run_atr(k) <= MAX_SINCE_TRIGGER_ATR:
                ema_cross = True
    return squeeze_release, aroon_turn, chop_break, ema_cross


def evaluate(df):
    """ERKEN TREND girisi. Sira: temel filtreler -> yon -> erken tetikler (en az 2) -> ZORUNLU hacim -> puan (0-100).
    ADX zorunlu degil: ADX_MIN altindaysa (olu piyasa) elenir, ustunde kademeli puan verir."""
    if len(df) < 60:
        return None, "veri_az"
    last = df.iloc[-1]
    needed = ["adx", "rsi", "atr", "plus_di", "minus_di", "ema_fast", "ema_slow", "macd_hist",
              "sq_mom", "aroon_up", "aroon_dn", "chop"]
    if (any(pd.isna(last[k]) for k in needed) or pd.isna(df["adx"].iloc[-4]) or pd.isna(df["macd_hist"].iloc[-2])
            or pd.isna(df["sq_mom"].iloc[-2]) or pd.isna(df["aroon_up"].iloc[-5]) or pd.isna(df["aroon_dn"].iloc[-5])
            or pd.isna(df["chop"].iloc[-1 - max(CHOP_LOOKBACK, CHOP_FALLING_BARS)])):
        return None, "veri_eksik"

    close = float(last["close"])
    atr = float(last["atr"])
    adx = float(last["adx"])
    rsi = float(last["rsi"])
    if close <= 0 or atr <= 0:
        return None, "veri_eksik"

    atr_ref = float(df["atr"].iloc[-ATR_REF_BARS:].mean())
    if max(atr, atr_ref) / close < MIN_ATR_PCT:      # hem su an hem gun ortalamasi durgunsa: komisyon yer, girme
        return None, "durgun"
    atr = max(atr, ATR_FLOOR_RATIO * atr_ref)        # sikisma bitiminde ATR kucuk: stop asiri dar olmasin
    if adx < ADX_MIN:
        return None, "adx_dusuk"
    adx_rising = adx > float(df["adx"].iloc[-4])
    if adx < ADX_FULL and not adx_rising:
        return None, "adx_dusuyor"     # ADX dusukse YUKSELIYOR olmali (trend doguyor); dusuyorsa trend olu

    # --- Yon: 3 gostergeden en az 2'si ayni yonde olmali ---
    ef, es = float(last["ema_fast"]), float(last["ema_slow"])
    mom = float(last["sq_mom"])
    mom_prev = float(df["sq_mom"].iloc[-2])
    # Squeeze momentumu yon degistirmekte gecikir; bu yuzden 3 gostergenin COGUNLUGU (momentum, Aroon, DI) yonu belirler.
    long_votes = int(mom > 0) + int(last["aroon_up"] > last["aroon_dn"]) + int(last["plus_di"] > last["minus_di"])
    short_votes = int(mom < 0) + int(last["aroon_dn"] > last["aroon_up"]) + int(last["minus_di"] > last["plus_di"])
    if long_votes >= 2 and long_votes > short_votes:
        side = "long"
    elif short_votes >= 2 and short_votes > long_votes:
        side = "short"
    else:
        return None, "yon_belirsiz"

    if side == "long" and rsi > RSI_MAX_LONG:
        return None, "rsi_asiri"
    if side == "short" and rsi < RSI_MIN_SHORT:
        return None, "rsi_asiri"
    # --- GEC KALMA FILTRELERI (anti-chase) ---
    if abs(close - es) / atr > ENTRY_MAX_EXT_ATR:
        return None, "gec_kalmis"          # fiyat EMA21'den cok uzak
    if len(df) >= MOVE_24H_BARS:
        w24 = df.iloc[-MOVE_24H_BARS:]
        if side == "long":
            base = float(w24["low"].min())
            if base > 0 and close / base - 1.0 > MAX_MOVE_24H:
                return None, "24s_asiri"   # long: 24s dibinden cok yukari kosmus
        else:
            top = float(w24["high"].max())
            if top > 0 and 1.0 - close / top > MAX_MOVE_24H:
                return None, "24s_asiri"   # short: 24s zirvesinden cok asagi kosmus
    if float(last["high"] - last["low"]) > MAX_CANDLE_ATR * atr:
        return None, "dikey_mum"           # dikey mum: geri cekilmeyi bekle

    # MACD teyidi (mevcut ayar): momentum sinyal yonunde ve zayiflamiyor olmali
    if MACD_FILTER_ENABLED:
        mh = float(last["macd_hist"])
        mh_prev = float(df["macd_hist"].iloc[-2])
        if side == "long" and not (mh > 0 and mh >= mh_prev):
            return None, "macd_zayif"
        if side == "short" and not (mh < 0 and mh <= mh_prev):
            return None, "macd_zayif"

    per_hour = max(int(3_600_000 // TF_MS), 1)
    if len(df) > per_hour:
        ref_close = float(df["close"].iloc[-1 - per_hour])
        if ref_close > 0 and abs(close / ref_close - 1.0) > MAX_MOVE_1H:
            return None, "pump_1s"

    # --- Erken tetikler: en az EARLY_MIN_TRIGGERS tanesi (4 tetikten) ---
    sq_rel, aroon_turn, chop_break, ema_cross = _trigger_states(df, side, atr)
    trig = [name for name, ok in (("squeeze", sq_rel), ("aroon", aroon_turn), ("chop", chop_break),
                                  ("ema", ema_cross)) if ok]
    if len(trig) < EARLY_MIN_TRIGGERS:
        return None, "erken_sinyal_yok"

    # --- ZORUNLU hacim sicramasi: son 2 mumun en yuksegi / onceki 20 mum ortalamasi ---
    vol_base = float(df["volume"].iloc[-22:-2].mean())
    v2 = df["volume"].iloc[-2:]
    spike = float(v2.max()) / vol_base if vol_base > 0 else 0.0
    if spike < VOL_SPIKE_MIN:
        return None, "hacim_dusuk"
    spike_candle = df.iloc[-2:].loc[v2.idxmax()]
    if (side == "long" and spike_candle["close"] < spike_candle["open"]) or \
       (side == "short" and spike_candle["close"] > spike_candle["open"]):
        return None, "hacim_ters"      # hacim ters yonlu mumda patlamis: yon teyidi degil

    # --- Puan (0-100; ust zaman dilimi puani sonradan eklenir) ---
    # EMA kesisimi sadece tetik olarak puanlanir (yon icin ayri EMA puani YOK: cift sayim olmasin)
    trig_pts = (12 if sq_rel else 0) + (12 if aroon_turn else 0) + (8 if chop_break else 0) + (8 if ema_cross else 0)   # 40
    vol_pts = 10 + 10 * clamp((spike - VOL_SPIKE_MIN) / (VOL_SPIKE_FULL - VOL_SPIKE_MIN), 0, 1)       # 10-20
    adx_pts = 15 * clamp((adx - ADX_MIN) / (ADX_FULL - ADX_MIN), 0, 1) + (5 if adx_rising else 0)      # 20
    mom_up = (mom > mom_prev) if side == "long" else (mom < mom_prev)
    mom_pts = (5 if mom_up else 0) + 5 * clamp(abs(mom) / atr, 0, 1)                                    # 10
    if side == "long":
        rsi_pts = 5 if 50 <= rsi <= 68 else (2 if 45 <= rsi <= RSI_MAX_LONG else 0)
        breakout = close > float(df["high"].iloc[-1 - BREAKOUT_BARS:-1].max())
    else:
        rsi_pts = 5 if 32 <= rsi <= 50 else (2 if RSI_MIN_SHORT <= rsi <= 55 else 0)
        breakout = close < float(df["low"].iloc[-1 - BREAKOUT_BARS:-1].min())
    breakout_pts = 5 if breakout else 0

    score = trig_pts + vol_pts + adx_pts + mom_pts + rsi_pts + breakout_pts
    return {
        "side": side, "score": score, "atr": atr, "price": close,
        "adx": adx, "rv": spike, "rsi": rsi, "htf": "-", "trig": "+".join(trig),
    }, "ok"


def evaluate_classic(df):
    """ESKI (gec ama saglam) trend degerlendirmesi: sadece ROTASYONDA 'pozisyonun trendi suruyor mu' diye bakmak icin."""
    if len(df) < 60:
        return None, "veri_az"
    last = df.iloc[-1]
    needed = ["adx", "rsi", "atr", "plus_di", "minus_di", "ema_fast", "ema_slow", "macd_hist"]
    if any(pd.isna(last[k]) for k in needed) or pd.isna(df["adx"].iloc[-4]) or pd.isna(df["macd_hist"].iloc[-2]):
        return None, "veri_eksik"

    close = float(last["close"])
    atr = float(last["atr"])
    adx = float(last["adx"])
    rsi = float(last["rsi"])
    if close <= 0 or atr <= 0:
        return None, "veri_eksik"

    if atr / close < MIN_ATR_PCT:
        return None, "durgun"
    if adx < ADX_CLASSIC_MIN:
        return None, "adx_dusuk"

    ef, es = float(last["ema_fast"]), float(last["ema_slow"])
    if ef > es and last["plus_di"] > last["minus_di"] and close > es:
        side = "long"
    elif ef < es and last["minus_di"] > last["plus_di"] and close < es:
        side = "short"
    else:
        return None, "yon_belirsiz"

    if side == "long" and rsi > RSI_MAX_LONG:
        return None, "rsi_asiri"
    if side == "short" and rsi < RSI_MIN_SHORT:
        return None, "rsi_asiri"
    if abs(close - es) / atr > MAX_EXT_ATR:
        return None, "gec_kalmis"

    # MACD teyidi: momentum sinyal yonunde ve zayiflamiyor olmali
    if MACD_FILTER_ENABLED:
        mh = float(last["macd_hist"])
        mh_prev = float(df["macd_hist"].iloc[-2])
        if side == "long" and not (mh > 0 and mh >= mh_prev):
            return None, "macd_zayif"
        if side == "short" and not (mh < 0 and mh <= mh_prev):
            return None, "macd_zayif"

    per_hour = max(int(3_600_000 // TF_MS), 1)
    if len(df) > per_hour:
        ref_close = float(df["close"].iloc[-1 - per_hour])
        if ref_close > 0 and abs(close / ref_close - 1.0) > MAX_MOVE_1H:
            return None, "pump_1s"          # son 1 saatte cok oynamis: pump/dump'a gec kalinmis olabilir

    vol_recent = float(df["volume"].iloc[-3:].mean())
    vol_base = float(df["volume"].iloc[-23:-3].mean())
    rv = vol_recent / vol_base if vol_base > 0 else 0.0
    if rv < MIN_REL_VOL:
        return None, "hacim_dusuk"

    adx_pts = 25 * clamp((adx - ADX_CLASSIC_MIN) / (40 - ADX_CLASSIC_MIN), 0, 1)
    rising_pts = 10 if adx > float(df["adx"].iloc[-4]) else 0
    spread_pts = 10 * clamp(abs(ef - es) / atr, 0, 1)
    vol_pts = 20 * clamp((rv - 0.8) / 1.2, 0, 1)
    if side == "long":
        rsi_pts = 10 if 50 <= rsi <= 68 else (5 if 45 <= rsi <= RSI_MAX_LONG else 0)
        breakout = close > float(df["high"].iloc[-1 - BREAKOUT_BARS:-1].max())
    else:
        rsi_pts = 10 if 32 <= rsi <= 50 else (5 if RSI_MIN_SHORT <= rsi <= 55 else 0)
        breakout = close < float(df["low"].iloc[-1 - BREAKOUT_BARS:-1].min())
    breakout_pts = 10 if breakout else 0

    score = adx_pts + rising_pts + spread_pts + vol_pts + rsi_pts + breakout_pts
    return {
        "side": side, "score": score, "atr": atr, "price": close,
        "adx": adx, "rv": rv, "rsi": rsi, "htf": "-",
    }, "ok"


htf_cache = {}   # symbol -> (zaman, df)   (sadece REST yedek modu)


def htf_alignment(symbol, side):
    df = None
    if ws_healthy() and is_fresh(to_raw(symbol)):
        df = htf_df_from_ws(symbol)          # REST yok: 15 dk mumlardan uretilen 1s mumlar
    if df is None:
        now = time.time()
        cached = htf_cache.get(symbol)
        if cached and now - cached[0] < HTF_CACHE_SECONDS:
            df = cached[1]
        else:
            wait_if_banned()
            df = fetch_df_rest(symbol, HTF)
            htf_cache[symbol] = (now, df)
    if len(df) < 30:
        return "belirsiz"
    last = df.iloc[-1]
    up = last["ema_fast"] > last["ema_slow"] and last["close"] > last["ema_slow"]
    down = last["ema_fast"] < last["ema_slow"] and last["close"] < last["ema_slow"]
    if side == "long":
        return "ok" if up else ("ters" if down else "belirsiz")
    return "ok" if down else ("ters" if up else "belirsiz")


def scan_market(skip):
    global last_scan
    wait_for_ws()
    symbols = get_dynamic_symbols()
    tradfi = {s for s in symbols if is_tradfi(s)}
    if tradfi:
        symbols = [s for s in symbols if s not in tradfi]
        names = ", ".join(sorted(s.split("/")[0] for s in tradfi)[:12])
        log.info(f"TradFi (hisse/emtia) vadeliler elendi: {len(tradfi)} adet ({names}...)")
    ws_ok = ws_healthy()
    log.info(f"Hacim araligindaki coin sayisi: {len(symbols)} | "
             f"veri kaynagi: {'WebSocket' if ws_ok else 'REST YEDEK'}")
    if not symbols:
        return []

    movers_view = []
    if ws_ok:
        movers = prescreen_movers(symbols)
        if movers is None:
            log.info(f"⏳ Isinma: fiyat gecmisi birikiyor (en az {PRESCREEN_WARMUP // 60} dk). Bu sure islem acilmaz.")
            last_scan = {"time": time.time(), "kaynak": "WebSocket (isinma)", "adaylar": [], "hareketli": []}
            return []
        hot = [m["symbol"] for m in movers if m["symbol"] not in skip]
        movers_view = [{"coin": m["symbol"], "degisim_yuzde": round(m["chg"], 2),
                        "hacim_x": round(m["ratio"], 1)} for m in movers]
        if movers:
            top = ", ".join(f"{m['symbol'].split('/')[0]} {m['chg']:+.1f}%" for m in movers[:8])
            log.info(f"🔥 Hareketli coinler ({len(movers)}): {top}")
        mark_hot(hot)
        prepare_data(hot, skip)       # sadece bu adaylarin akisi acilir ve gecmis mumu (bir kez) yuklenir
        hot_set = set(hot)
        symbols = hot + [s for s in symbols if s not in hot_set and is_fresh(to_raw(s))]
    else:
        symbols = symbols[:FALLBACK_MAX_COINS]
        log.warning(f"⚠️ WebSocket hazir degil: REST yedek modu, sadece ilk {len(symbols)} coin, yavas taranacak.")

    reasons = {}
    stage1 = []

    for symbol in symbols:
        if symbol in skip:
            continue
        try:
            if ws_ok:
                if not is_fresh(to_raw(symbol)):
                    reasons["veri_yok"] = reasons.get("veri_yok", 0) + 1
                    continue
                df = get_df(symbol)
                if df is None:
                    reasons["veri_yok"] = reasons.get("veri_yok", 0) + 1
                    continue
            else:
                wait_if_banned()
                df = fetch_df_rest(symbol, TIMEFRAME)
                time.sleep(FALLBACK_SLEEP)
        except Exception as e:
            log.error(f"{symbol} veri hatasi: {e}")
            if note_ban(e):
                return []
            continue
        cand, reason = evaluate(df)
        if cand is None:
            reasons[reason] = reasons.get(reason, 0) + 1
            continue
        if (cand["side"] == "long" and not ALLOW_LONG) or (cand["side"] == "short" and not ALLOW_SHORT):
            continue
        cand["symbol"] = symbol
        stage1.append(cand)

    stage1.sort(key=lambda c: c["score"], reverse=True)
    final = []
    # BTC sadece REFERANS: coin onun yonune/hizina gore ne kadar guclu diye bakilir, BTC'de islem acilmaz. REST yok (WS gecmisi).
    btc_ref = btc_change_pct() if (ws_ok and REL_STRENGTH_ENABLED) else None
    for cand in stage1[:HTF_CHECK_TOP]:
        try:
            htf = htf_alignment(cand["symbol"], cand["side"])
        except Exception as e:
            log.error(f"{cand['symbol']} ust zaman hatasi: {e}")
            if note_ban(e):
                break
            continue
        if htf == "ters":
            reasons["ust_trend_ters"] = reasons.get("ust_trend_ters", 0) + 1
            continue
        cand["score"] += HTF_PTS if htf == "ok" else HTF_PTS / 2
        cand["htf"] = htf
        rs_pts, rs_diff = relative_strength(cand["symbol"], cand["side"], btc_ref)
        cand["score"] += rs_pts
        cand["rs"] = rs_diff
        final.append(cand)

    final.sort(key=lambda c: c["score"], reverse=True)

    # OPEN INTEREST teyidi: sadece en iyi OI_CHECK_TOP aday (esigi gecebilecek olanlar), onbellekli tek REST istegi, fail-open.
    if OI_ENABLED and ws_ok and final:
        oi_pool = [c for c in final if c["score"] >= MIN_TREND_SCORE - OI_PTS][:OI_CHECK_TOP]
        for c in oi_pool:
            chg = oi_change_pct(c["symbol"])
            c["oi"] = chg
            if chg is None:
                continue
            if chg >= OI_RISE_PCT:
                c["score"] += OI_PTS
            elif chg <= -OI_DROP_PCT:
                c["score"] -= OI_PTS
        final.sort(key=lambda c: c["score"], reverse=True)

    log.info(f"Analiz edilen: {len(symbols)} | Elenenler: {reasons} | Trend adayi: {len(stage1)} | "
             f"Ust trend teyidi gecen: {len(final)}")
    for c in final[:5]:
        rs_txt = f"{c['rs']:+.2f}" if c.get("rs") is not None else "-"
        oi_txt = f"{c['oi']:+.2f}%" if c.get("oi") is not None else "-"
        log.info(f"  ⭐ {c['symbol']} {c['side'].upper()} puan={c['score']:.0f} adx={c['adx']:.0f} "
                 f"hacimx={c['rv']:.1f} rsi={c['rsi']:.0f} 1s={c['htf']} tetik={c.get('trig', '-')} "
                 f"btc_fark={rs_txt} oi={oi_txt}")
    last_scan = {
        "time": time.time(),
        "kaynak": "WebSocket" if ws_ok else "REST yedek",
        "hareketli": movers_view,
        "adaylar": [{"coin": c["symbol"], "yon": c["side"], "puan": round(c["score"]),
                     "adx": round(c["adx"]), "hacim_x": round(c["rv"], 1), "rsi": round(c["rsi"]),
                     "ust_trend": c["htf"], "tetik": c.get("trig", "-"),
                     "btc_fark_yuzde": round(c["rs"], 2) if c.get("rs") is not None else None,
                     "oi_degisim_yuzde": round(c["oi"], 2) if c.get("oi") is not None else None}
                    for c in final[:10]],
    }

    ready = [c for c in final if c["score"] >= MIN_TREND_SCORE]
    if BTC_FILTER_ENABLED and ws_ok and ready:
        btc = btc_change_pct()
        if btc is not None:
            kept = [c for c in ready
                    if not ((c["side"] == "long" and btc <= -BTC_BLOCK_PCT)
                            or (c["side"] == "short" and btc >= BTC_BLOCK_PCT))]
            if len(kept) < len(ready):
                log.info(f"🛡️ BTC filtresi: BTC son {BTC_WINDOW // 60} dk %{btc:+.2f}, "
                         f"ters yondeki {len(ready) - len(kept)} aday elendi.")
            ready = kept
    return ready


# ---------------- KORUMA EMIRLERI ----------------
def sl_distance(price, atr, lev):
    """Stop mesafesi: SL_ATR x ATR, ama marjinde SL_MAX_MARGIN_PCT zarari gecmeyecek kadar (hangisi yakinsa)."""
    cap = price * (SL_MAX_MARGIN_PCT / 100.0) / max(float(lev), 1.0)
    return min(SL_ATR * atr, cap)


def calc_tp_amount(symbol, amount):
    """Ilk kar alda kapanacak miktar = pozisyonun TP1_CLOSE_PCT'si.
    Yuvarlama yuzunden miktar yariyi belirgin asarsa (kucuk pozisyonda 'yarisi' = tamami olabilir) 0 doner: TP konmaz,
    cunku yanlis buyuk bir kar-al pozisyonun tamamini kapatir. Stop-loss yine de aktif kalir."""
    try:
        tp_amount = float(ex.amount_to_precision(symbol, amount * TP1_CLOSE_PCT))
    except Exception as e:
        log.warning(f"{symbol} kar-al miktari hesaplanamadi: {e}")
        return 0.0
    if tp_amount <= 0 or tp_amount > amount * (TP1_CLOSE_PCT + 0.10):
        return 0.0
    return tp_amount


def place_tp_once(symbol, exit_side, tp_amount, tp_price):
    """Kar-al emrini TAM BIR KERE koyar.
    1) Koymadan once borsada bu yonde zaten kar-al var mi bakar; varsa yenisini KOYMAZ.
    2) Istek zaman asimina ugrarsa emir yine de gitmis olabilir: tekrar gondermeden once borsaya bakar.
    3) Koyduktan sonra sayar; birden fazla gorurse fazlasini iptal eder.
    Basarili ise kalan kar-al emrinin id'sini, degilse None dondurur."""
    existing = scan_exit_orders(symbol, exit_side)[1]
    if existing:
        log.warning(f"⚠️ {symbol}: borsada zaten {len(existing)} kar-al emri var, yeni kar-al KONMADI.")
        return enforce_single_tp(symbol, exit_side)

    try:
        ex.create_order(symbol, "TAKE_PROFIT_MARKET", exit_side, tp_amount, None,
                        {"stopPrice": tp_price, "reduceOnly": True})
    except Exception as e:
        log.error(f"{symbol} TP gonderilirken hata: {e}")
        note_ban(e)
        time.sleep(2)                      # zaman asimiysa emir gitmis olabilir: kontrol et, KORUMASIZCA tekrar gonderme
        if not scan_exit_orders(symbol, exit_side)[1]:
            return None
        log.warning(f"⚠️ {symbol}: hata alindi ama kar-al emri borsada gorunuyor, tekrar gonderilmedi.")

    time.sleep(1)                          # borsa listesine yansisin
    return enforce_single_tp(symbol, exit_side)


def enforce_single_tp(symbol, exit_side):
    """Borsada bu yonde birden fazla kar-al varsa EN YENISI kalir (id en buyuk), digerleri iptal edilir.
    Kural sabit oldugu icin iki bot kopyasi calissa bile ikisi de ayni emri tutar, birbirinin emrini silip
    kar-alsiz birakmaz. Kalan kar-al emrinin id'sini dondurur (yoksa None)."""
    tps = scan_exit_orders(symbol, exit_side)[1]
    if not tps:
        return None
    tps.sort(key=lambda x: str(x[0]).zfill(20))
    keep_id = tps[-1][0]
    if len(tps) > 1:
        removed = sum(1 for oid, _ in tps[:-1] if cancel_safe(oid, symbol))
        log.warning(f"🧹 {symbol}: {len(tps)} kar-al emri bulundu (iki kar-al = pozisyonun TAMAMI kapanir!), "
                    f"{removed} fazla iptal edildi. Birden fazla bot kopyasi calisiyor olabilir!")
    return keep_id


def place_protection(symbol, entry, amount, atr, side="long", lev=1):
    """SL: giris -/+ sl_distance (en fazla SL_ATR x ATR, marjinde en fazla %SL_MAX_MARGIN_PCT zarar).
    TP1: marjine gore TP1_MARGIN_PCT kar (fiyat hareketi = pct / kaldirac), pozisyonun TP1_CLOSE_PCT'si, TAM 1 emir."""
    exit_side = "sell" if side == "long" else "buy"
    tp_move = (TP1_MARGIN_PCT / 100.0) / max(lev, 1)      # fiyatin yuzde kac oynamasi gerek
    sl_dist = sl_distance(entry, atr, lev)
    if side == "long":
        sl_raw = entry - sl_dist
        tp_raw = entry * (1 + tp_move)
    else:
        sl_raw = entry + sl_dist
        tp_raw = entry * (1 - tp_move)
    sl_price = float(ex.price_to_precision(symbol, sl_raw))
    tp_price = float(ex.price_to_precision(symbol, tp_raw))
    tp_amount = calc_tp_amount(symbol, amount)

    try:
        sl_order = ex.create_order(symbol, "STOP_MARKET", exit_side, amount, None,
                                   {"stopPrice": sl_price, "reduceOnly": True})
        log.info(f"🛑 SL kondu: {symbol} @ {sl_price}")
    except Exception as e:
        log.error(f"{symbol} SL konamadi: {e}")
        return None

    tp_id = None
    if tp_amount > 0:
        tp_id = place_tp_once(symbol, exit_side, tp_amount, tp_price)
        if tp_id is not None:
            log.info(f"🎯 TP kondu (%{TP1_CLOSE_PCT*100:.0f} kapanir, marjinde %{TP1_MARGIN_PCT:.0f}): "
                     f"{symbol} @ {tp_price} miktar={tp_amount}")
        else:
            log.error(f"{symbol} TP konamadi")
    else:
        log.warning(f"{symbol}: kar-al miktari uygun degil (yuvarlama), TP konmadi. SL aktif.")
    return {"sl_id": sl_order.get("id"), "sl_price": sl_price,
            "tp_id": tp_id, "tp_price": tp_price, "tp_amount": tp_amount}


def cancel_safe(order_id, symbol):
    last_err = None
    for params in ({}, {"trigger": True}):
        try:
            ex.cancel_order(order_id, symbol, params)
            return True
        except Exception as e:
            last_err = e
            continue
    log.warning(f"{symbol} emir iptal edilemedi (id={order_id}): {last_err}")
    return False


def cancel_all_safe(symbol):
    for params in ({}, {"trigger": True}):
        try:
            ex.cancel_all_orders(symbol, params)
        except Exception:
            pass


def _raw_type(o):
    """Emrin Binance'teki gercek tipi (ccxt'nin birlestirilmis 'type' alani STOP_MARKET'i 'market' gosterebilir)."""
    info = o.get("info") or {}
    for k in ("orderType", "origType", "type"):
        v = info.get(k)
        if v:
            return str(v).upper()
    return str(o.get("type") or "").upper()


def scan_exit_orders(symbol, exit_side):
    """Borsadaki bu yone ait cikis emirlerini TEK taramada okur.
    Dondurur: ([(id, fiyat) stop-loss], [(id, fiyat) kar-al], {id: miktar})."""
    stops, tps, amounts = {}, {}, {}
    for params in ({}, {"trigger": True}):
        try:
            orders = ex.fetch_open_orders(symbol, params=params)
        except Exception as e:
            log.warning(f"{symbol} acik emirler okunamadi {params}: {e}")
            note_ban(e)
            continue
        for o in orders:
            if o.get("side") != exit_side:
                continue
            info = o.get("info") or {}
            raw = _raw_type(o)
            is_tp = raw.startswith("TAKE_PROFIT")
            if not (is_tp or raw.startswith("STOP") or o.get("stopLossPrice")):
                continue
            price = (o.get("stopLossPrice") or o.get("stopPrice") or o.get("triggerPrice")
                     or info.get("stopPrice") or info.get("triggerPrice"))
            try:
                price = float(price)
            except (TypeError, ValueError):
                continue
            if price <= 0 or o.get("id") is None:
                continue
            (tps if is_tp else stops)[o["id"]] = price
            try:
                amt = o.get("amount")
                if amt is None:
                    amt = info.get("origQty") or info.get("quantity")
                if amt is not None:
                    amounts[o["id"]] = float(amt)
            except (TypeError, ValueError):
                pass
    return list(stops.items()), list(tps.items()), amounts


def find_stop_orders(symbol, exit_side):
    """Borsadaki bu yone ait STOP-LOSS emirlerini [(id, fiyat), ...] verir (kar al emirleri sayilmaz)."""
    return scan_exit_orders(symbol, exit_side)[0]


def dedupe_stops(symbol):
    """Her dakika calisir. (1) KAR-AL: pozisyon basina en fazla 1 kar-al; ilk kar-al dolduysa hic kalmasin;
    kar-al miktari yariyi asiyorsa duzelt. (2) STOP-LOSS: birden fazla varsa en siki olani tut, fazlasini iptal et."""
    with trail_lock:
        st = trail_state.get(symbol)
        if not st:
            return
        exit_side = "sell" if st["side"] == "long" else "buy"
        tp_state = st.get("tp_state", "pending")
        orig = float(st.get("orig_amount") or 0)
        cur = float(st.get("amount") or 0)
    stops, tps, amounts = scan_exit_orders(symbol, exit_side)      # REST, kilidin disinda (tek tarama)

    # ---------- KAR-AL EMIRLERI ----------
    # Pozisyon ilk miktarin %90'inin altina indiyse ilk kar-al gerceklesmistir.
    if tp_state != "done" and orig > 0 and 0 < cur < orig * 0.9:
        tp_state = "done"
        with trail_lock:
            cu = trail_state.get(symbol)
            if cu:
                cu["tp_state"] = "done"
                save_state()
        log.info(f"🎯 {symbol}: ilk kar-al gerceklesti ({orig} -> {cur}); bundan sonra hic kar-al emri kalmayacak.")

    if tp_state == "done":
        if tps:
            removed_tp = sum(1 for oid, _ in tps if cancel_safe(oid, symbol))
            log.warning(f"🧹 {symbol}: ilk kar-al tamamlandi ama {len(tps)} kar-al emri kalmisti "
                        f"(kalan yariyi kapatirdi), {removed_tp} tanesi iptal edildi.")
    else:
        if len(tps) > 1:
            tps.sort(key=lambda x: str(x[0]).zfill(20))        # id'ler artar: en yeniyi tut
            removed_tp = sum(1 for oid, _ in tps[:-1] if cancel_safe(oid, symbol))
            log.warning(f"🧹 {symbol}: {len(tps)} kar-al emri bulundu (iki kar-al = pozisyonun TAMAMI kapanir!), "
                        f"{removed_tp} fazla kar-al iptal edildi. Birden fazla bot kopyasi calisiyor olabilir!")
            tps = tps[-1:]
        if tps and orig > 0:
            # Tek kalan kar-al da cok buyukse (yariyi asiyorsa) pozisyonun fazlasini kapatir: dogru miktarla yeniden koy
            oid, tpx = tps[0]
            amt = amounts.get(oid)
            expected = calc_tp_amount(symbol, orig)
            if amt is not None and expected > 0 and amt > expected * 1.2:
                log.warning(f"🧹 {symbol}: kar-al miktari {amt}, olmasi gereken {expected}. Duzeltiliyor.")
                if cancel_safe(oid, symbol):
                    place_tp_once(symbol, exit_side, expected, tpx)

    # ---------- STOP-LOSS EMIRLERI ----------
    if not stops:
        restore_missing_stop(symbol)         # borsada hic stop yok: yenisini koy (konamazsa kapat)
        return
    with trail_lock:
        cs = trail_state.get(symbol)
        if cs:
            cs["nostop_n"] = 0
    if len(stops) <= 1:
        return
    keep_id, keep_price = (max if exit_side == "sell" else min)(stops, key=lambda x: x[1])
    removed = 0
    for oid, _ in stops:
        if oid != keep_id and cancel_safe(oid, symbol):
            removed += 1
    with trail_lock:
        cur_st = trail_state.get(symbol)
        if cur_st:
            cur_st["sl_id"] = keep_id
            cur_st["sl_price"] = keep_price
            save_state()
    log.info(f"🧹 {symbol}: {len(stops)} stop bulundu, {removed} fazla stop iptal edildi, kalan SL: {keep_price}")


def adopt_position(symbol, pos):
    side = pos.get("side")
    size = float(pos.get("contracts") or 0)
    entry = float(pos.get("entryPrice") or 0)
    if side not in ("long", "short") or size <= 0 or entry <= 0:
        return False
    df = fetch_df_rest(symbol, TIMEFRAME)
    atr = float(df.iloc[-1]["atr"]) if len(df) else 0.0
    if not atr or atr != atr:
        return False
    price = float(ex.fetch_ticker(symbol)["last"])
    exit_side = "sell" if side == "long" else "buy"
    stops, tps, _ = scan_exit_orders(symbol, exit_side)
    log.info(f"{symbol}: borsada {len(stops)} stop-loss, {len(tps)} kar-al emri bulundu")
    sl_id, sl_price = None, None
    if stops:
        # en yakin (en siki) stopu tut, fazlalari temizle
        sl_id, sl_price = (max if exit_side == "sell" else min)(stops, key=lambda x: x[1])
        extra = [oid for oid, _ in stops if oid != sl_id]
        for oid in extra:
            cancel_safe(oid, symbol)
        if extra:
            log.info(f"🧹 {symbol}: {len(extra)} fazla stop temizlendi")
    if sl_id is None:
        try:
            lev_pos = float(pos.get("leverage") or LEV_NORMAL)
        except (TypeError, ValueError):
            lev_pos = float(LEV_NORMAL)
        d = sl_distance(price, atr, lev_pos)
        raw = price - d if side == "long" else price + d
        sl_price = float(ex.price_to_precision(symbol, raw))
        amount = float(ex.amount_to_precision(symbol, size))
        order = ex.create_order(symbol, "STOP_MARKET", exit_side, amount, None,
                                {"stopPrice": sl_price, "reduceOnly": True})
        sl_id = order.get("id")
        log.warning(f"🛟 {symbol} stop'suz pozisyon bulundu, SL kondu @ {sl_price}")
    ext = max(entry, price) if side == "long" else min(entry, price)
    opened_at = time.time()
    try:
        ut = (pos.get("info") or {}).get("updateTime")      # borsanin son guncelleme zamani (yaklasik pozisyon yasi)
        if ut:
            opened_at = min(opened_at, float(ut) / 1000.0)
    except (TypeError, ValueError):
        pass
    with trail_lock:
        try:
            lev = float(pos.get("leverage") or LEV_NORMAL)
        except (TypeError, ValueError):
            lev = float(LEV_NORMAL)
        size_amt = float(ex.amount_to_precision(symbol, size))
        trail_state[symbol] = {
            "side": side, "atr": atr, "ext": ext, "entry": entry, "sl_price": sl_price,
            "sl_id": sl_id, "amount": size_amt, "lev": lev,
            # Borsada kar-al varsa ilk kar-al henuz dolmamis demektir (1 tane kalir, fazlasi temizlenir).
            # Yoksa ya doldu ya hic konmadi: bundan sonra yeni kar-al KONMAZ, rastgele gelen kar-al iptal edilir.
            "orig_amount": size_amt, "tp_state": "pending" if tps else "done",
            "opened_at": opened_at,
        }
        save_state()
    log.info(f"🤝 {symbol} ({side}) pozisyonu sahiplenildi, gezen stop surecek. SL: {sl_price}")
    return True


# ---------------- DARALAN GEZEN STOP (TRAILING) ----------------
def trail_multiplier(profit_atr):
    """Kar (ATR cinsinden) buyudukce takip mesafesi (ATR carpani) daralir."""
    for limit, mult in TRAIL_TIERS:
        if profit_atr < limit:
            return mult
    return TRAIL_TIGHT_ATR


def update_trailing_stop_logic(symbol, price):
    """Anlik fiyatla gezen stopu hesaplar. REST'e SADECE stop gercekten tasinacaksa gider.
    Kar TRAIL_START_ATR'ye ulasmadan stop yerinde kalir (sadece ilk stop-loss). Sonra takip mesafesi
    kar buyudukce daralir (TRAIL_TIERS -> TRAIL_TIGHT_ATR) ve hicbir zaman fiyatin %TRAIL_MIN_DIST_PCT'inden az olmaz.
    Stop en az TRAIL_MIN_STEP_ATR ilerlemeden ve son guncellemenin uzerinden TRAIL_MIN_INTERVAL sn gecmeden REST'e gidilmez.
    Paylasilan state trail_lock altinda okunur/yazilir; REST cagrilari kilidin DISINDA yapilir."""
    with trail_lock:
        st = trail_state.get(symbol)
        if not st:
            return

        is_long = st["side"] == "long"
        exit_side = "sell" if is_long else "buy"
        entry = float(st["entry"])
        atr = st["atr"]

        if is_long:
            if price <= st["ext"]:
                return
            st["ext"] = max(st["ext"], price)
            profit_atr = (st["ext"] - entry) / atr
            if profit_atr < TRAIL_START_ATR:
                return
            mult = trail_multiplier(profit_atr)
            dist = max(mult * atr, st["ext"] * TRAIL_MIN_DIST_PCT / 100.0)
            raw_sl = st["ext"] - dist
            if profit_atr >= BREAKEVEN_AT_ATR:
                raw_sl = max(raw_sl, entry * (1 + FEE_BUFFER_PCT / 100.0))
            new_sl = float(ex.price_to_precision(symbol, raw_sl))
            move_ok = new_sl > st["sl_price"] + TRAIL_MIN_STEP_ATR * atr and new_sl < price
        else:
            if price >= st["ext"]:
                return
            st["ext"] = min(st["ext"], price)
            profit_atr = (entry - st["ext"]) / atr
            if profit_atr < TRAIL_START_ATR:
                return
            mult = trail_multiplier(profit_atr)
            dist = max(mult * atr, st["ext"] * TRAIL_MIN_DIST_PCT / 100.0)
            raw_sl = st["ext"] + dist
            if profit_atr >= BREAKEVEN_AT_ATR:
                raw_sl = min(raw_sl, entry * (1 - FEE_BUFFER_PCT / 100.0))
            new_sl = float(ex.price_to_precision(symbol, raw_sl))
            move_ok = new_sl < st["sl_price"] - TRAIL_MIN_STEP_ATR * atr and new_sl > price

        if not move_ok or time.time() - st.get("last_move_ts", 0) < TRAIL_MIN_INTERVAL:
            return
        st["last_move_ts"] = time.time()
        old_sl_id = st["sl_id"]
        old_sl_price = st["sl_price"]
        amount_now = st.get("amount")

    # ---- kilit disinda: REST islemleri ----
    wait_if_banned()
    try:
        # Kismi kar alindiysa pozisyon kuculmustur: stopu GUNCEL miktarla koy (sadece stop tasirken 1 istek)
        try:
            for p in ex.fetch_positions([symbol]):
                if p.get("symbol") == symbol:
                    c = float(p.get("contracts") or 0)
                    if c > 0:
                        amount_now = float(ex.amount_to_precision(symbol, c))
                    break
        except Exception as e:
            log.warning(f"{symbol} guncel pozisyon miktari okunamadi, eski miktar kullanilacak: {e}")
            note_ban(e)

        new_order = ex.create_order(symbol, "STOP_MARKET", exit_side, amount_now, None,
                                    {"stopPrice": new_sl, "reduceOnly": True})
        cancel_safe(old_sl_id, symbol)
        log.info(f"📈 {symbol} gezen stop: {old_sl_price} -> {new_sl} (kar {profit_atr:.1f} ATR, takip {mult} ATR)")
        with trail_lock:
            cur = trail_state.get(symbol)
            if cur is st:
                cur["sl_id"] = new_order.get("id")
                cur["sl_price"] = new_sl
                cur["amount"] = amount_now
                save_state()
    except Exception as e:
        log.error(f"{symbol} gezen stop guncelleme hatasi: {e}")
        note_ban(e)


def sync_positions(symbols):
    """TEK REST istegiyle: kapanan pozisyonlari temizle, kalan miktari guncelle."""
    wait_if_banned()
    positions = ex.fetch_positions()
    open_map = {}
    for p in positions:
        c = float(p.get("contracts") or 0)
        if c > 0:
            open_map[p["symbol"]] = c
    for symbol in symbols:
        if symbol in open_map:
            with trail_lock:
                st = trail_state.get(symbol)
                if st:
                    st["amount"] = float(ex.amount_to_precision(symbol, open_map[symbol]))
            try:
                dedupe_stops(symbol)          # her dakika: fazladan kar-al/stop kaldiysa temizle
            except Exception as e:
                log.warning(f"{symbol} fazla emir temizligi hatasi: {e}")
                note_ban(e)
        else:
            cancel_all_safe(symbol)
            with trail_lock:
                closed_st = trail_state.pop(symbol, None)
                cooldown[symbol] = time.time() + COOLDOWN_SECONDS
                save_state()
            log.info(f"{symbol} kapandi, {COOLDOWN_SECONDS // 60} dk bekleme basladi.")
            journal_close(symbol, closed_st, "borsada kapandi (stop / kar-al / gezen stop)")


# ---------------- YEDEK STOP (borsadaki stop calismazsa bot kendisi kapatir) ----------------
def close_tracked_position(symbol, neden, journal_reason):
    """Takip edilen pozisyonu market emriyle kapatir, emirleri siler, durumu temizler, gunluge yazar."""
    with trail_lock:
        st0 = trail_state.get(symbol)
        if not st0:
            return False
        side = st0["side"]
    wait_if_banned()
    amount = None
    try:
        for p in ex.fetch_positions([symbol]):
            if p.get("symbol") == symbol:
                c = float(p.get("contracts") or 0)
                if c > 0:
                    amount = float(ex.amount_to_precision(symbol, c))
                break
    except Exception as e:
        log.error(f"{symbol} {neden}: pozisyon miktari okunamadi: {e}")
        note_ban(e)
        return False
    if not amount:
        return False                # zaten kapanmis: sync_positions temizler
    if not emergency_close(symbol, side, amount, neden=neden):
        return False
    cancel_all_safe(symbol)
    with trail_lock:
        closed_st = trail_state.pop(symbol, None)
        cooldown[symbol] = time.time() + COOLDOWN_SECONDS
        save_state()
    journal_close(symbol, closed_st, journal_reason)
    return True


def restore_missing_stop(symbol):
    """Borsada hic stop-loss yoksa (iptal olmus, hic konmamis vb.) yenisini koyar; konamazsa pozisyonu kapatir.
    dedupe_stops icinden dakikada bir cagrilir. Emir listesi okunamadiysa yanlis alarm olmasin diye
    ust uste 2 kontrolde stop gorunmezse harekete gecer."""
    with trail_lock:
        st = trail_state.get(symbol)
        if not st:
            return
        st["nostop_n"] = st.get("nostop_n", 0) + 1
        n = st["nostop_n"]
        side, entry, atr = st["side"], float(st["entry"]), float(st["atr"])
        lev = float(st.get("lev") or LEV_NORMAL)
        amount = float(st.get("amount") or 0)
    if n < 2:
        return
    exit_side = "sell" if side == "long" else "buy"
    try:
        price = float(ex.fetch_ticker(symbol)["last"])
        d = sl_distance(entry, atr, lev)
        # Stop girisin degil, mevcut fiyatin dogru tarafinda olmali (fiyat stopu coktan gectiyse hemen kapat)
        sl_raw = entry - d if side == "long" else entry + d
        already_past = price <= sl_raw if side == "long" else price >= sl_raw
        if already_past:
            log.warning(f"🆘 {symbol}: stop yoktu ve fiyat stop seviyesini gecmis, pozisyon kapatiliyor.")
            close_tracked_position(symbol, "stop yok + fiyat gecmis", "stop yoktu, fiyat stopu gecmisti")
            return
        sl_price = float(ex.price_to_precision(symbol, sl_raw))
        pos_amt = amount
        for p in ex.fetch_positions([symbol]):
            if p.get("symbol") == symbol and float(p.get("contracts") or 0) > 0:
                pos_amt = float(ex.amount_to_precision(symbol, float(p["contracts"])))
                break
        order = ex.create_order(symbol, "STOP_MARKET", exit_side, pos_amt, None,
                                {"stopPrice": sl_price, "reduceOnly": True})
        with trail_lock:
            cur = trail_state.get(symbol)
            if cur:
                cur["sl_id"] = order.get("id")
                cur["sl_price"] = sl_price
                cur["nostop_n"] = 0
                save_state()
        log.warning(f"🛟 {symbol}: borsada stop yoktu, yeni SL kondu @ {sl_price}")
    except Exception as e:
        log.error(f"{symbol} eksik stop yenilenemedi: {e}")
        note_ban(e)
        log.warning(f"🆘 {symbol}: stop konamadi, pozisyon kapatiliyor.")
        close_tracked_position(symbol, "stop konamadi", "stop yoktu ve konamadi")


def backup_stop_check(symbol, price):
    """Borsadaki stop calismadiysa / cok kaydiysa: fiyat stopu BACKUP_STOP_GAP_PCT kadar gectiginde
    pozisyonu bot kendisi market emriyle kapatir. Kontrol WS fiyatiyla yapilir (REST'e SADECE tetiklenince gider)."""
    with trail_lock:
        st = trail_state.get(symbol)
        if not st:
            return
        side = st["side"]
        sl = float(st["sl_price"])
        if time.time() - st.get("backup_ts", 0) < 20:       # ust uste tekrar denemesin
            return
    gap = BACKUP_STOP_GAP_PCT / 100.0
    breach = price <= sl * (1 - gap) if side == "long" else price >= sl * (1 + gap)
    if not breach:
        return
    with trail_lock:
        st["backup_ts"] = time.time()
    log.warning(f"🛟 {symbol} fiyat stopu gecti ({price} / SL {sl}), yedek stop kapatiyor.")
    close_tracked_position(symbol, "yedek stop", "yedek stop (borsa stopu calismadi)")


def trail_loop():
    log.info("Gezen stop dongusu basladi (WS fiyati, REST sadece stop tasirken).")

    # Ilk acilista mevcut pozisyonlari sahiplen
    try:
        wait_if_banned()
        for p in ex.fetch_positions():
            c = float(p.get("contracts") or 0)
            if c > 0:
                symbol = p["symbol"]
                with trail_lock:
                    known = symbol in trail_state
                if not known:
                    adopt_position(symbol, p)
    except Exception as e:
        log.error(f"Baslangic pozisyon sahiplenme hatasi: {e}")
        note_ban(e)

    last_sync = 0.0        # acilista hemen bir temizlik turu: restart sonrasi kalan fazla kar-al/stop silinsin
    while True:
        interval = TRAIL_TICK_SECONDS
        try:
            wait_if_banned()
            with trail_lock:
                symbols = list(trail_state.keys())

            if symbols:
                prices = {}
                if ws_healthy():
                    with ws_lock:
                        for s in symbols:
                            t = tickers.get(to_raw(s))
                            if t:
                                prices[s] = t["price"]
                else:
                    # WS yok: yedek olarak tek istekle fiyatlar, yavas aralikla
                    interval = TRAIL_CHECK_SECONDS
                    tk = ex.fetch_tickers(symbols)
                    for s in symbols:
                        if s in tk and tk[s].get("last"):
                            prices[s] = float(tk[s]["last"])

                for symbol, price in prices.items():
                    try:
                        backup_stop_check(symbol, price)
                        update_trailing_stop_logic(symbol, price)
                    except Exception as e:
                        log.error(f"{symbol} gezen stop hatasi: {e}")
                        note_ban(e)

                if time.time() - last_sync >= POSITION_SYNC_SECONDS:
                    last_sync = time.time()
                    sync_positions(symbols)
        except Exception as e:
            log.error(f"Gezen stop dongu hatasi: {e}")
            note_ban(e)
            interval = TRAIL_CHECK_SECONDS

        time.sleep(interval)

def emergency_close(symbol, side, amount, neden="korumasiz pozisyon"):
    """Pozisyonu market emriyle (reduceOnly) kapatir; 3 kez dener. Basarili mi diye True/False doner."""
    close_side = "sell" if side == "long" else "buy"
    for attempt in range(1, 4):
        try:
            wait_if_banned()
            ex.create_order(symbol, "market", close_side, amount, None, {"reduceOnly": True})
            log.warning(f"🔒 {symbol} kapatildi ({neden}, deneme {attempt}).")
            return True
        except Exception as e:
            log.error(f"{symbol} kapatma hatasi ({neden}, deneme {attempt}/3): {e}")
            note_ban(e)
            time.sleep(2)
    return False


def min_notional(symbol):
    """Borsanin bu coin icin en kucuk emir degeri (USDT). Bilgi yoksa 5 varsayilir."""
    try:
        m = (ex.markets or {}).get(symbol) or {}
        c = ((m.get("limits") or {}).get("cost") or {}).get("min")
        return float(c) if c else 5.0
    except Exception:
        return 5.0


def open_trade(cand, open_symbols):
    symbol, side, atr = cand["symbol"], cand["side"], cand["atr"]
    lev = LEV_STRONG if cand["score"] >= STRONG_SCORE else LEV_NORMAL

    if not TRADING_ENABLED:
        # KAGIT MODU: emir verilmez, sadece sinyal ve olasi stop/kar-al seviyeleri loglanir
        d = sl_distance(cand["price"], atr, lev)
        sl_p = cand["price"] - d if side == "long" else cand["price"] + d
        mv = (TP1_MARGIN_PCT / 100.0) / max(lev, 1)
        tp_p = cand["price"] * (1 + mv) if side == "long" else cand["price"] * (1 - mv)
        log.info(f"📝 [KAGIT MODU] {side.upper()} {symbol} puan={cand['score']:.0f} fiyat={cand['price']} "
                 f"SL={sl_p:.8g} TP1={tp_p:.8g} tetik={cand.get('trig', '-')} adx={cand.get('adx', 0):.0f} "
                 f"hacimx={cand.get('rv', 0):.1f} (emir verilmedi)")
        journal("signal", symbol, side=side, score=round(cand["score"]), lev=lev, entry=cand["price"],
                sl=round(sl_p, 10), tp=round(tp_p, 10), reason=f"kagit modu (emir verilmedi) tetik={cand.get('trig', '-')} adx={cand.get('adx', 0):.0f} hacimx={cand.get('rv', 0):.1f}")
        with trail_lock:
            cooldown[symbol] = time.time() + 30 * 60        # ayni sinyal her turda tekrar loglanmasin
            save_state()
        return False

    with open_lock:
        opening_symbols.add(symbol)
    try:
        # Bu coinde pozisyon yok (tarama bunu garanti eder): onceki islemden kalan artik stop/kar-al emirleri varsa temizle,
        # yoksa yeni pozisyona yanlislikla eski emirler (ornegin ikinci bir kar-al) baglanir.
        cancel_all_safe(symbol)
        try:
            ex.set_margin_mode("isolated", symbol)
        except Exception:
            pass

        # Kaldirac: once hedef (15x), coin desteklemezse sirayla daha dusuk degerler denenir.
        # Gercekten ayarlanan kaldirac kullanilir; hic biri ayarlanamazsa islem acilmaz
        # (aksi halde bot 15x varsayip stopu yanlis hesaplardi).
        lev_ok = None
        for try_lev in (lev,) + LEV_FALLBACKS:
            try:
                ex.set_leverage(try_lev, symbol)
                lev_ok = try_lev
                break
            except Exception as e:
                log.warning(f"{symbol} {try_lev}x kaldirac ayarlanamadi: {e}")
                note_ban(e)
        if lev_ok is None:
            log.warning(f"⏭️ {symbol}: kaldirac ayarlanamadi, islem atlandi.")
            return False
        if lev_ok != lev:
            log.info(f"{symbol}: {lev}x desteklenmiyor, {lev_ok}x kullanilacak.")
        lev = lev_ok

        # ---- RISK BAZLI POZISYON BOYUTU ----
        # Stop olursa kaybedilecek tutar = miktar x stop mesafesi = hesabin RISK_PCT'si olacak sekilde miktar bulunur.
        # Marjin ust siniri (MARGIN_PCT) yine gecerli: hangisi kucukse o kullanilir.
        bal = ex.fetch_balance()["USDT"]
        usdt_free = float(bal["free"])
        usdt_total = float(bal["total"])
        sl_dist = sl_distance(cand["price"], atr, lev)
        if sl_dist <= 0 or usdt_total <= 0:
            return False
        risk_usdt = usdt_total * RISK_PCT / 100.0
        amount_risk = risk_usdt / sl_dist
        amount_cap = usdt_free * MARGIN_PCT * lev / cand["price"]
        amount = float(ex.amount_to_precision(symbol, min(amount_risk, amount_cap)))
        if amount <= 0:
            return False
        min_cost = min_notional(symbol)
        if amount * cand["price"] < min_cost * 1.02:
            log.warning(f"⏭️ {symbol}: hesaplanan pozisyon degeri {amount * cand['price']:.2f} USDT, borsa en az "
                        f"{min_cost:.0f} USDT istiyor, bu islem atlandi.")
            return False
        real_risk = amount * sl_dist
        log.info(f"⚖️ {symbol} boyut: risk hedefi {risk_usdt:.2f} USDT, stop mesafesi %{sl_dist / cand['price'] * 100:.2f}, "
                 f"stop olursa ~{real_risk:.2f} USDT kayip, marjin ~{amount * cand['price'] / lev:.2f} USDT")

        if side == "long":
            order = ex.create_market_buy_order(symbol, amount)
        else:
            order = ex.create_market_sell_order(symbol, amount)
        entry = float(order.get("average") or order.get("price") or cand["price"])
        log.info(f"🚀 ISLEM ACILDI ({side.upper()}): {symbol} | Puan: {cand['score']:.0f} | "
                 f"Miktar: {amount} | Kaldirac: {lev}x | Giris: {entry}")
        open_symbols.append(symbol)

        prot = place_protection(symbol, entry, amount, atr, side, lev)
        if prot:
            with trail_lock:
                trail_state[symbol] = {
                    "side": side, "atr": atr, "ext": entry, "entry": entry,
                    "sl_price": prot["sl_price"], "sl_id": prot["sl_id"], "amount": amount,
                    "lev": lev,
                    "orig_amount": amount,
                    "tp_state": "pending" if prot.get("tp_id") is not None else "done",
                    "opened_at": time.time(),
                }
                save_state()
            journal("open", symbol, side=side, score=round(cand["score"]), lev=lev, entry=entry, amount=amount,
                    sl=prot["sl_price"], tp=prot["tp_price"],
                    risk_usdt=round(amount * abs(entry - prot["sl_price"]), 4))
        else:
            log.error(f"{symbol} SL yok, pozisyon kapatiliyor!")
            if emergency_close(symbol, side, amount):
                if symbol in open_symbols:
                    open_symbols.remove(symbol)
            else:
                log.critical(f"🆘 {symbol} pozisyonu KORUMASIZ ve KAPATILAMADI! Sahiplenme (SL koyma) deneniyor. "
                             f"Gerekirse ELLE mudahale et!")
                try:
                    adopt_position(symbol, {"side": side, "contracts": amount, "entryPrice": entry,
                                            "leverage": lev})
                except Exception as e:
                    log.critical(f"🆘 {symbol} sahiplenme de basarisiz: {e} -> HEMEN ELLE KAPAT!")
        return True
    finally:
        with open_lock:
            opening_symbols.discard(symbol)

def position_pnl_margin_pct(st, price):
    """Pozisyonun marjine gore anlik kar/zarar yuzdesi (komisyon haric). Veri yoksa None."""
    try:
        entry = float(st.get("entry") or 0)
        if entry <= 0 or price <= 0:
            return None
        lev = float(st.get("lev") or LEV_NORMAL)
        move = (price - entry) / entry if st["side"] == "long" else (entry - price) / entry
        return move * lev * 100.0
    except (TypeError, ValueError, KeyError):
        return None


def own_trend_score(symbol, side):
    """Pozisyondaki coinin KENDI yonundeki trend puani (WS onbellekten, REST yok). Trend yoksa/veri yoksa 0."""
    try:
        raw = to_raw(symbol)
        if not is_fresh(raw):
            return 0.0
        df = get_df(symbol)
        if df is None:
            return 0.0
        cand, _ = evaluate_classic(df)
        if cand and cand["side"] == side:
            return float(cand["score"])
    except Exception:
        pass
    return 0.0


def find_stale_positions():
    """Bayat pozisyonlar: yeterince eski + marjine gore kar/zarar -/+ bantin icinde + kendi yonunde guclu trend kalmamis.
    REST istegi atmaz. Dondurur: [(coin, kar%, yas_dk, kendi_puani)], en bayat basta."""
    if not ws_healthy():
        return []
    now = time.time()
    with trail_lock:
        snap = {s: dict(st) for s, st in trail_state.items()}
    stale = []
    for symbol, st in snap.items():
        age = now - float(st.get("opened_at") or now)
        if age < ROTATE_MIN_AGE_SECONDS:
            continue
        with ws_lock:
            t = tickers.get(to_raw(symbol))
        if not t:
            continue
        pnl = position_pnl_margin_pct(st, t["price"])
        if pnl is None or not (ROTATE_PNL_MIN <= pnl <= ROTATE_PNL_MAX):
            continue
        own = own_trend_score(symbol, st["side"])
        if own >= ROTATE_KEEP_SCORE:
            continue
        stale.append((symbol, pnl, age / 60.0, own))
    stale.sort(key=lambda x: (x[3], -x[2]))       # once trendi en zayif, sonra en eski
    return stale


def close_position_for_rotation(symbol):
    """Pozisyonu market emriyle kapatir, kalan emirleri siler, durumu temizler. Basarili ise True."""
    with trail_lock:
        st = trail_state.get(symbol)
        if not st:
            return False
        side = st["side"]
    wait_if_banned()
    amount = None
    try:
        for p in ex.fetch_positions([symbol]):
            if p.get("symbol") == symbol:
                c = float(p.get("contracts") or 0)
                if c > 0:
                    amount = float(ex.amount_to_precision(symbol, c))
                break
    except Exception as e:
        log.error(f"{symbol} rotasyon: pozisyon miktari okunamadi: {e}")
        note_ban(e)
        return False
    if not amount:
        return False                      # zaten kapanmis: sync_positions temizler
    if not emergency_close(symbol, side, amount, neden="rotasyon"):
        return False
    cancel_all_safe(symbol)
    with trail_lock:
        closed_st = trail_state.pop(symbol, None)
        cooldown[symbol] = time.time() + COOLDOWN_SECONDS
        save_state()
    journal_close(symbol, closed_st, "rotasyon (bayat pozisyon)")
    return True


def rotate_if_needed(open_symbols):
    """Pozisyon limiti doluyken calisir. Bayat pozisyon varsa ve guclu (>= ROTATE_MIN_SCORE) bir trend bulunursa
    bayati kapatip yeni coine girer. Rotasyon yapildiysa True doner."""
    if not ROTATE_ENABLED or not TRADING_ENABLED:
        return False
    now = time.time()
    with rotate_lock:
        rotate_times[:] = [t for t in rotate_times if now - t < 86400]
        if rotate_times and now - rotate_times[-1] < ROTATE_GAP_SECONDS:
            return False
        if len(rotate_times) >= ROTATE_MAX_PER_DAY:
            return False

    stale = find_stale_positions()
    if not stale:
        return False
    names = ", ".join(f"{s.split('/')[0]} (%{p:+.1f}, {a:.0f} dk, puan {o:.0f})" for s, p, a, o in stale)
    log.info(f"💤 Bayat pozisyon(lar): {names}. Daha guclu trend araniyor...")

    if daily_loss_hit():
        return False

    with trail_lock:
        skip = set(open_symbols) | set(cooldown.keys())
    candidates = scan_market(skip)
    strong = [c for c in candidates if c["score"] >= ROTATE_MIN_SCORE]
    if not strong:
        log.info(f"Rotasyon icin yeterince guclu trend yok (esik {ROTATE_MIN_SCORE}), bayat pozisyon korunuyor.")
        return False

    cand = strong[0]
    symbol, pnl, age_min, own = stale[0]
    log.info(f"🔄 ROTASYON: {symbol} (%{pnl:+.1f}, {age_min:.0f} dk, puan {own:.0f}) kapatilip "
             f"{cand['symbol']} {cand['side'].upper()} (puan {cand['score']:.0f}) acilacak.")
    if not close_position_for_rotation(symbol):
        log.warning(f"{symbol} rotasyon icin kapatilamadi, yeni islem acilmadi.")
        return False
    if symbol in open_symbols:
        open_symbols.remove(symbol)
    with rotate_lock:
        rotate_times.append(time.time())
    try:
        open_trade(cand, open_symbols)
    except Exception as e:
        log.error(f"{cand['symbol']} rotasyon sonrasi islem acma hatasi: {e}")
        note_ban(e)
    return True


def bot_loop():
    log.info("Bot arkaplan dongusu basladi.")
    if not TRADING_ENABLED:
        log.warning("📝 KAGIT MODU: TRADING_ENABLED=0, bot emir vermeyecek, sadece sinyalleri loglayacak.")
    while True:
        wait_if_banned()
        try:
            try:
                open_symbols = get_open_symbols()
            except Exception as e:
                log.error(f"Pozisyonlar cekilemedi: {e}")
                note_ban(e)
                time.sleep(30)
                continue

            if len(open_symbols) >= MAX_OPEN:
                log.info(f"Pozisyon limiti dolu ({len(open_symbols)}/{MAX_OPEN}), yeni tarama yapilmadi.")
                try:
                    rotate_if_needed(open_symbols)
                except Exception as e:
                    log.error(f"Rotasyon hatasi: {e}")
                    note_ban(e)
                time.sleep(LOOP_SECONDS)
                continue

            if daily_loss_hit():
                time.sleep(LOOP_SECONDS)
                continue

            now = time.time()
            with trail_lock:
                for s in [s for s, t in cooldown.items() if t <= now]:
                    cooldown.pop(s, None)
                skip = set(open_symbols) | set(cooldown.keys())

            candidates = scan_market(skip)
            if not candidates:
                log.info("Bu turda esigi gecen trend yok, islem acilmadi.")

            for cand in candidates:
                if len(open_symbols) >= MAX_OPEN:
                    break
                if cand["symbol"] in open_symbols:
                    continue
                try:
                    open_trade(cand, open_symbols)
                    time.sleep(2)
                except Exception as inner_err:
                    log.error(f"{cand['symbol']} islem hatasi: {inner_err}")
                    if "-4411" in str(inner_err):
                        # TradFi-Perps sozlesmesi imzalanmamis (hisse/emtia vadelisi): 24 saat atla
                        with trail_lock:
                            cooldown[cand["symbol"]] = time.time() + 24 * 3600
                            save_state()
                        log.warning(f"🚫 {cand['symbol']} TradFi-Perps sozlesmesi gerektiriyor, 24 saat atlanacak.")
                        continue
                    if note_ban(inner_err):
                        break

        except Exception as e:
            log.error(f"Dongu hatasi: {e}")
            note_ban(e)

        time.sleep(LOOP_SECONDS)


_started = False

    global _started
    with _start_lock:
        if not _started:
            _started = True

            threading.Thread(target=ws_thread, daemon=True).start()
            threading.Thread(target=bot_loop, daemon=True).start()
            threading.Thread(target=trail_loop, daemon=True).start()


start_bot_once()

@app.route("/scan")
def scan_view():
    """Hareketli coinler + son tarama adaylari + acik pozisyonlarin durumu (kar %, yas, rotasyona aday mi)."""
    now = time.time()
    with trail_lock:
        snap = {s: dict(st) for s, st in trail_state.items()}
    stale_names = {s for s, _, _, _ in find_stale_positions()}
    pozisyonlar = []
    for s, st in snap.items():
        with ws_lock:
            t = tickers.get(to_raw(s))
        pnl = position_pnl_margin_pct(st, t["price"]) if t else None
        pozisyonlar.append({
            "coin": s,
            "yon": st.get("side"),
            "kar_marjin_yuzde": round(pnl, 2) if pnl is not None else None,
            "yas_dk": round((now - float(st.get("opened_at") or now)) / 60),
            "bayat": s in stale_names,
        })
    btc = btc_change_pct()
    return jsonify({
        "son_tarama_sn_once": round(now - last_scan["time"]) if last_scan["time"] else None,
        "tarama_kaynagi": last_scan["kaynak"],
        "btc_son_15dk_yuzde": round(btc, 2) if btc is not None else None,
        "hareketli": last_scan.get("hareketli", []),
        "adaylar": last_scan["adaylar"],
        "pozisyonlar": pozisyonlar,
        "rotasyon_acik": ROTATE_ENABLED,
        "bugunku_rotasyon": len([t for t in rotate_times if now - t < 86400]),
    })


@app.route("/journal")
def journal_view():
    """Kapanan islemlerin ozeti: islem sayisi, kazanma orani, net kar/zarar, islem basi beklenen deger + son kayitlar."""
    rows = []
    try:
        with open(JOURNAL_FILE, newline="") as f:
            rows = list(csv.DictReader(f))
    except Exception:
        pass
    nets = []
    for r in rows:
        if r.get("event") == "close" and r.get("net_usdt") not in (None, ""):
            try:
                nets.append(float(r["net_usdt"]))
            except ValueError:
                pass
    wins = [n for n in nets if n > 0]
    losses = [n for n in nets if n <= 0]
    return jsonify({
        "kapanan_islem": len(nets),
        "kazanan": len(wins),
        "kazanma_orani_yuzde": round(len(wins) / len(nets) * 100, 1) if nets else None,
        "net_toplam_usdt": round(sum(nets), 3) if nets else None,
        "islem_basi_net_usdt": round(sum(nets) / len(nets), 4) if nets else None,
        "ort_kazanc_usdt": round(sum(wins) / len(wins), 4) if wins else None,
        "ort_kayip_usdt": round(sum(losses) / len(losses), 4) if losses else None,
        "son_kayitlar": rows[-30:],
    })
